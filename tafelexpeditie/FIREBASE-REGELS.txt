service cloud.firestore {
  match /databases/{database}/documents {

Plaats het VOOR de afsluitende match /{document=**}.

    function isGekoppeldAanKlas(schooljaar, klasId) {
      return request.auth != null
        && exists(/databases/$(database)/documents/klasleerkrachten/$(schooljaar + "_" + klasId))
        && (
          request.auth.uid in get(/databases/$(database)/documents/klasleerkrachten/$(schooljaar + "_" + klasId)).data.leerkracht_uids
          || request.auth.token.email in get(/databases/$(database)/documents/klasleerkrachten/$(schooljaar + "_" + klasId)).data.leerkracht_emails
        );
    }

    function isGekoppeldAanGroep(groep) {
      return groep.groepId == "graad1"
          && (isGekoppeldAanKlas(groep.schooljaar, "1A") || isGekoppeldAanKlas(groep.schooljaar, "2A"))
        || groep.groepId == "graad2"
          && (isGekoppeldAanKlas(groep.schooljaar, "3A") || isGekoppeldAanKlas(groep.schooljaar, "4A"))
        || groep.groepId == "graad3"
          && (isGekoppeldAanKlas(groep.schooljaar, "5A") || isGekoppeldAanKlas(groep.schooljaar, "6A"));
    }

    match /schoolbeheer_groepen/{groepDocId} {
      allow read, update: if request.auth != null
        && (isSchoolBreed() || isSecretariaat() || isGekoppeldAanGroep(resource.data));

      allow create: if isSecretariaat();
      allow delete: if isBeheerder();
    }

    // Lichte leerlingtoegang voor TafelExpeditie.
    // Een QR bevat een lange, willekeurige documentsleutel. Lijsten opvragen is
    // verboden; alleen een exact gekende QR of leerlingcode kan gelezen worden.
    match /tafelexpeditie_klassen/{toegangscode} {
      allow get: if request.auth != null;
      allow list: if false;

      allow create: if request.auth != null
        && (
          isSchoolBreed()
          || isSecretariaat()
          || isGekoppeldAanKlas(
            request.resource.data.schoolyear,
            request.resource.data.className
          )
        );

      allow update, delete: if request.auth != null
        && (
          isSchoolBreed()
          || isSecretariaat()
          || isGekoppeldAanKlas(
            resource.data.schoolyear,
            resource.data.className
          )
        );

      match /leerlingen/{geheimeLeerlingSleutel} {
        allow get: if request.auth != null;
        allow list: if false;

        allow create: if request.auth != null
          && (
            isSchoolBreed()
            || isSecretariaat()
            || isGekoppeldAanKlas(
              request.resource.data.schoolyear,
              request.resource.data.className
            )
          );

        allow delete: if request.auth != null
          && (
            isSchoolBreed()
            || isSecretariaat()
            || isGekoppeldAanKlas(
              resource.data.schoolyear,
              resource.data.className
            )
          );

        allow update: if request.auth != null
          && (
            request.resource.data.diff(resource.data).affectedKeys()
              .hasOnly(['sessions', 'factStats', 'completedAssignmentIds'])
            || isSchoolBreed()
            || isSecretariaat()
            || isGekoppeldAanKlas(
              resource.data.schoolyear,
              resource.data.className
            )
          );
      }

      // Tijdelijke live klasgames. De lange QR-sleutel en geheime
      // leerlingdocumentsleutel werken als toegangssleutels. Leerlingen mogen
      // nooit de deelnemerslijst opvragen; alleen de leerkracht ziet het bord.
      match /races/{raceId} {
        allow get: if request.auth != null;
        allow list: if false;

        allow create, update, delete: if request.auth != null
          && (
            isSchoolBreed()
            || isSecretariaat()
            || isGekoppeldAanKlas(
              get(/databases/$(database)/documents/tafelexpeditie_klassen/$(toegangscode)).data.schoolyear,
              get(/databases/$(database)/documents/tafelexpeditie_klassen/$(toegangscode)).data.className
            )
          );

        match /racers/{geheimeLeerlingSleutel} {
          allow get: if request.auth != null;

          allow list: if request.auth != null
            && (
              isSchoolBreed()
              || isSecretariaat()
              || isGekoppeldAanKlas(
                get(/databases/$(database)/documents/tafelexpeditie_klassen/$(toegangscode)).data.schoolyear,
                get(/databases/$(database)/documents/tafelexpeditie_klassen/$(toegangscode)).data.className
              )
            );

          allow create: if request.auth != null
            && request.resource.data.keys().hasOnly([
              'classNumber', 'firstName', 'progress', 'total', 'wrong',
              'finished', 'joinedAt', 'updatedAt', 'finishedAt', 'elapsedMs'
            ]);

          allow update: if request.auth != null
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
              'progress', 'total', 'wrong', 'finished', 'updatedAt',
              'finishedAt', 'elapsedMs'
            ]);

          allow delete: if request.auth != null
            && (
              isSchoolBreed()
              || isSecretariaat()
              || isGekoppeldAanKlas(
                get(/databases/$(database)/documents/tafelexpeditie_klassen/$(toegangscode)).data.schoolyear,
                get(/databases/$(database)/documents/tafelexpeditie_klassen/$(toegangscode)).data.className
              )
            );
        }
      }
    }
